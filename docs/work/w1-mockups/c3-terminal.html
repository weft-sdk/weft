Express, KafkaJS, Socket.ioserver.ts
const app = express()
app.use(express.json())
const io = new Server(createServer(app))
const producer = new Kafka({ brokers }).producer()
await producer.connect()

app.post("/orders", async (req, res) => {
  const order = await orders.create(req.body)  // any
  await producer.send({ topic: "order.placed",
    messages: [{ key: order.id, value: JSON.stringify(order) }] })
  io.to(`order:${order.id}`).emit("order.status", "placed")
  res.json(order)
})
Weftserver.ts
weft.http.post("/orders", async ({ body }) => {  // CreateOrder
  const order = await orders.create(body)
  await weft.kafka.publish("order.placed", order)
  weft.ws.to(`order:${order.id}`).emit("order.status", "placed")
  return order
})

await weft.start()

Write the type once. Every service gets it.

A contract names each event, route and call with its type. weft gen turns it into typed code for every service, in its own language.

package orders @version("1.0.0")

type Money {
  amount: int            // minor units
  currency: Currency
}

type Order {
  id: uuid
  customerId: string
  items: OrderItem[]
  total: Money
  placedAt: timestamp
}

event order.placed: Order {
  key: id
}
