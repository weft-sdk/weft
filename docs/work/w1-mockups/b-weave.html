    Before and after
    
      
    The same endpoint, written twice.
    
      
    Take an order over HTTP, publish it to Kafka, tell the customer's browser over a socket. Today that means three libraries, three setups and a request body typed as any.
    
    
    
    
    
      
    
        
    Express, KafkaJS and Socket.ioserver.ts
    

    import express from "express"
import { createServer } from "node:http"
import { Kafka } from "kafkajs"
import { Server } from "socket.io"

const app = express()
app.use(express.json())
const server = createServer(app)
const io = new Server(server)

const kafka = new Kafka({ clientId: "orders", brokers: [process.env.KAFKA_BROKER] })
const producer = kafka.producer()
await producer.connect()

app.post("/orders", async (req, res) => {  const order = await orders.create(req.body)   // req.body: any  await producer.send({    topic: "order.placed",    messages: [{ key: order.id, value: JSON.stringify(order) }],  })  io.to(`order:${order.id}`).emit("order.status", "placed")  res.json(order)})
server.listen(3000)
    
      
    
      
    
        
    Weftserver.ts
    

    import { weft } from "./weft.gen"

weft.http.post("/orders", async ({ body }) => {   // body: CreateOrder  const order = await orders.create(body)  await weft.kafka.publish("order.placed", order)   // key from the contract  weft.ws.to(`order:${order.id}`).emit("order.status", "placed")  return order                                    // must be Order})
await weft.start()   // port, brokers and socket path come from weft.yaml
    
      
    
    
    
  
    

  
    
    
    
      
    Contracts
    
      
    Write the type once. Every language gets it.
    
      
    A .weft file names each event, route and call with its type. weft gen turns it into code for each service, so a Go consumer cannot drift from a TypeScript producer.
    
    
    
    
    
      
    
        
    contracts/orders.weft
    

    package orders @version("1.0.0")

type Money {
  amount: int        // minor units
  currency: Currency
}

type Order {
  id: uuid
  customerId: string
  items: OrderItem[]
  total: Money
  placedAt: timestamp
}

event order.placed: Order {
  key: id
}
